+function aAttendanceAnswer(t, ctx) {
+  return aAttendance().then(function (data) {
+    if (!data) return { text: "Attendance is taken on the online website: open the Attendance page, press Start attendance and show the QR code. Students scan it with their phone." };
+    if (!data.sessions.length) return { text: "You haven't taken attendance yet. Open the Attendance page, press Start attendance and show the QR code to your class." };
+    const course = aCourseIn(t, ctx);
+    const forCourse = data.sessions.filter(function (s) { return !course || s.course_id === course.id; });
+    const latest = forCourse[0] || data.sessions[0];
+    const rows = data.rows.filter(function (row) { return row.session_id === latest.id; });
+    const names = rows.map(function (row) { return nameFromEmail(row.email); });
+    const courseObj = findById(COURSES, latest.course_id) || course;
+    const group = courseObj ? aGroupFor(courseObj, ctx) : ctx.mine;
+    const absent = group.filter(function (r) { return !rows.some(function (row) { return String(row.email).toLowerCase() === r.email.toLowerCase(); }); });
+    const parts = ["Latest attendance for " + latest.course_name + " (" + aTimeText(latest.started_at) + (latest.ended_at ? "" : ", still open") + "): " + rows.length + " checked in."];
+    if (names.length) parts.push("Checked in:\n" + aList(names));
+    if (group.length) parts.push(absent.length ? "Not checked in (" + absent.length + " of the " + group.length + " students in that group):\n" + aList(absent.slice(0, 25).map(function (r) { return r.name; })) : "Everyone in that group checked in.");
+    if (data.sessions.length > 1 && group.length) {
+      const rates = data.sessions.map(function (s) {
+        const c = findById(COURSES, s.course_id);
+        const size = c ? aGroupFor(c, ctx).length : 0;
+        return size ? data.rows.filter(function (row) { return row.session_id === s.id; }).length / size : null;
+      }).filter(function (x) { return x !== null; });
+      if (rates.length) parts.push("Across your " + data.sessions.length + " attendance sessions, on average " + Math.round(Math.min(1, mean(rates)) * 100) + "% of the group checked in.");
+    }
+    return { text: parts.join("\n\n") };
+  });

+      if (!options.length) continue;
+      const when = offset * 1440 + options[0].start;
+      if (!best || when < best.when) best = { when: when, course: c, date: date, option: options[0], offset: offset };
