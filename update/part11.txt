+      break;
+    }
+  });
+  if (!best) return { text: "I couldn't work out your next class from the timetable." };
+  const n = aPickedCount(best.course, best.option.i);
+  const dayText = best.offset === 0 ? "today" : (best.offset === 1 ? "tomorrow" : aDateText(best.date));
+  return { text: (best.offset > 0 && t.indexOf("today") !== -1 ? "You have no class today. " : "") + "Your next class: " + best.course.name + ", " + dayText + " at " + aSlotText(best.course, best.option.i) + ", " + (best.course.room || TIMETABLE.defaultRoom) + ".\n\n" +
+    n + " student" + (n === 1 ? " has" : "s have") + " chosen this time." + (TIMETABLE.classDaysTbc ? " (Class days are still provisional.)" : "") };
+}

+function aLessonPlan(t, ctx) {
+  const course = aCourseIn(t, ctx);
+  if (!course) return { text: "You have no courses on the timetable yet." };
+  const weakest = aSkillAverages(ctx.mine)[0];
+  const focus = aSkillIn(t) || (weakest ? weakest.skill : null) || aCourseSkill(course) || SURVEY_SKILLS[0];
+  const own = aCourseSkill(course) || focus;
+  const tips = ASSISTANT_KNOWLEDGE.tips[focus.id] || [];
+  const warm = (ASSISTANT_KNOWLEDGE.warmups[own.id] || [])[0] || "";
+  const length = classLength(course);
+  const mark = function (fraction) { return Math.round(length * fraction); };
+  const due = allAssignments().find(function (a) { return a.course === course.id && isPublished(a); });
+  return { kind: "draft", text: "Lesson plan: " + course.name + " (" + levelById(course.level).name + " group, " + length + " minutes)\n" +
+    "Focus skill: " + focus.name + (weakest && weakest.skill.id === focus.id ? " (your group's weakest skill)" : "") + "\n\n" +
+    aList([
+      "0–" + mark(0.1) + " min · Warm-up: “" + warm + "”",
+      mark(0.1) + "–" + mark(0.28) + " min · Recap the last class and share today's goal",
+      mark(0.28) + "–" + mark(0.67) + " min · Main activity: " + (tips[0] || "guided practice on the main topic"),
+      mark(0.67) + "–" + mark(0.89) + " min · Practice in pairs: " + (tips[1] || "short tasks with feedback"),
+      mark(0.89) + "–" + length + " min · Wrap-up: one exit question" + (due ? ", and a reminder that “" + due.title + "” is due " + due.due : ""),
+    ]) };

+function aSummaryAnswer(ctx) {
+  const lines = [];
+  const avg = mean(ctx.mine.map(function (r) { return r.average; }));
+  const weakest = aSkillAverages(ctx.mine)[0];
+  lines.push("Your " + ctx.groupName + " group: " + ctx.mine.length + " student" + (ctx.mine.length === 1 ? "" : "s") + " with results" + (ctx.mine.length ? ", average " + aNum(avg) + " / " + ctx.max : ""));
+  if (weakest) lines.push("Skill to work on most: " + weakest.skill.name + " (" + aNum(weakest.average) + ")");
+  const pending = ctx.all.filter(function (r) { return !r.completed; }).length;
+  if (pending) lines.push(pending + " student" + (pending === 1 ? " is" : "s are") + " still doing the survey");
+  const ids = ctx.courses.map(function (c) { return c.id; });
+  const published = allAssignments().filter(function (a) { return ids.indexOf(a.course) !== -1 && isPublished(a); });
+  if (published.length) lines.push("Assignments listed: " + published.map(function (a) { return a.title + " (due " + a.due + ")"; }).join("; "));
+  return aAttendance().then(function (data) {
+    if (data && data.sessions.length) {
+      const latest = data.sessions[0];
+      const count = data.rows.filter(function (row) { return row.session_id === latest.id; }).length;
+      lines.push("Latest attendance: " + count + " checked in to " + latest.course_name + " (" + aTimeText(latest.started_at) + ")");
+    }
+    return { text: "Your summary:\n\n" + aList(lines) + "\n\nAsk me about any of these for more detail." };
+  });
+}

+// Chooses an answer: the first topic that matches the question.
+function assistantReply(text, prof) {
+  const t = aNorm(text);
+  const ctx = aContext(prof);
+  const intent = ASSISTANT_INTENTS.find(function (i) { return i.test(t, ctx); });
+  if (intent) return intent.answer(t, ctx);
+  const skill = aSkillIn(t);
+  if (skill) {
+    const avg = aSkillAverages(ctx.mine).find(function (x) { return x.skill.id === skill.id; });
+    return { text: skill.name + (avg ? ": your group averages " + aNum(avg.average) + " / " + ctx.max + "." : ".") + "\n\nTeaching ideas:\n" + aList(ASSISTANT_KNOWLEDGE.tips[skill.id] || []) };
+  }
+  const level = aLevelIn(t);
+  if (level) return ASSISTANT_INTENTS.find(function (i) { return i.id === "performance"; }).answer(" " + level.id + " ", ctx);
+  if (aUsernameIn(t)) return aStudentAnswer(t, ctx);
+  return { text: "I'm not sure I understood that. You could ask:\n\n" + aList(["How is my group doing?", "Who attended my last class?", "Tell me about ibz4000", "When is my next class?", "Give me teaching tips for writing"]) + "\n\nType “help” to see everything I can answer." };
+}
