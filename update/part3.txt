+      if (res.error) return;
+      const lists = {};
+      ids.forEach(function (id) { lists[id] = []; });
+      (res.data || []).forEach(function (r) { (lists[r.session_id] = lists[r.session_id] || []).push(r); });
+      att.lists = lists;
+      attDrawLists();
+    }, function () { /* offline: try again on the next tick */ });
+}

+function attDrawLists() {
+  Object.keys(att.sessions).concat(Object.keys(att.ended)).forEach(function (courseId) {
+    const s = att.sessions[courseId] || att.ended[courseId];
+    const list = att.lists[s.id] || [];
+    const count = document.getElementById("att-count-" + courseId);
+    const ul = document.getElementById("att-list-" + courseId);
+    if (count) count.textContent = fill(TEXT.attendance.countLine, { count: list.length });
+    if (ul) { ul.textContent = ""; addChildren(ul, attListItems(list)); }
+  });

+// Every second: redraw any QR code whose time window has changed, and the countdown.
+function attTick() {
+  if (!onAttendancePage()) return;
+  const now = Math.floor(Date.now() / 1000);
+  const w = attWindow();
+  const left = ATTENDANCE.codeSeconds - (now % ATTENDANCE.codeSeconds);
+  Object.keys(att.sessions).forEach(function (courseId) {
+    const s = att.sessions[courseId];
+    const timer = document.getElementById("att-timer-" + courseId);
+    if (timer) timer.textContent = fill(TEXT.attendance.changesIn, { seconds: left });
+    const box = document.getElementById("att-qr-" + courseId);
+    if (!box || box.getAttribute("data-window") === String(w)) return;
+    box.setAttribute("data-window", String(w));
+    attCode(s.secret, w).then(function (code) {
+      const svg = attQrSvg(attLink(s.id, code));
+      if (svg) {
+        box.innerHTML = svg;
+        const el = box.querySelector("svg");
+        if (el) { el.setAttribute("width", "100%"); el.setAttribute("height", "100%"); el.setAttribute("aria-hidden", "true"); }
+      } else {
+        box.textContent = TEXT.attendance.qrMissing;
+      }
+    });
+  });
+  if (now % ATTENDANCE.refreshSeconds === 0) attPoll();
+}

+// The professor's Attendance page.
+function staffAttendance(sec, ctx) {
+  const words = TEXT.attendance;
+  if (!attendanceSupported()) return sectionWrap(sec, null, h("p", { class: "notice notice-info" }, words.onlineOnly));
+  if (!att.loaded) attLoad();
+  if (!att.timer) att.timer = window.setInterval(attTick, 1000);
+  window.setTimeout(function () { attTick(); attDrawLists(); }, 0);
+  const courses = ctx.courses;
+  return sectionWrap(sec, null,
+    h("p", { class: "muted" }, fill(words.intro, { seconds: ATTENDANCE.codeSeconds })),
+    att.error ? h("p", { class: "notice notice-warning", role: "status" }, att.error) : null,
+    !att.loaded ? h("p", { class: "muted", role: "status" }, words.loading)
+      : (courses.length ? h("div", { class: "card-grid" }, courses.map(attCard)) : h("p", { class: "muted" }, words.noCourses))
+  );
